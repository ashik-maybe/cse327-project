# DB Schema + Constraints

users: id(uuid PK), name, email(unique), password_hash, role(volunteer|team_lead), created_at, updated_at.
cases: id(uuid PK), client_id(text, nullable), created_by(uuid FK), assigned_to(uuid FK nullable), summary(not null), description, category, urgency(low|medium|high|critical), status(new|in_progress|referred|resolved), location_name, latitude, longitude, next_contact, source(manual|voice), ai_assisted(bool), ai_source, client_created_at, updated_at. Unique: (created_by, client_id).
case_notes: id(uuid PK), case_id(uuid FK), user_id(uuid FK), note, created_at.
safety_reports: id(uuid PK), client_id(text nullable), created_by(uuid FK nullable), incident_type, anonymized_description, original_description(nullable), store_original(bool), approximate_location, urgency, status(new|acknowledged|resolved), occurred_at, created_at. Unique: (created_by, client_id).
service_points: id(uuid PK), name, category, area, description, contact_note, latitude, longitude, opening_hours, is_active(bool default true), created_at.
